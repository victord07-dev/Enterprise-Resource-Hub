✅ 1.6 Base Lot View
CREATE VIEW stock_lot_base AS
SELECT
  grni.id,
  grni.product_id,
  grni.grn_id,
  g.grn_number,
  g.received_date,
  grni.received_quantity,
  grni.buying_price,
  g.delivery_cost
FROM goods_receipt_note_items grni
JOIN goods_receipt_notes g ON g.id = grni.grn_id
WHERE g.status = 'confirmed';
