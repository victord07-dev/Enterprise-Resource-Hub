SELECT 
  ca.name,
  ca.opening_balance::numeric AS opening,
  
  (SELECT COALESCE(SUM(amount), 0) FROM customer_payments 
   WHERE cash_account_id = ca.id) AS customer_payments,
  
  (SELECT COALESCE(SUM(amount), 0) FROM payments 
   WHERE cash_account_id = ca.id 
   AND status = 'completed') AS sales_payments,
  
  (SELECT COALESCE(SUM(amount), 0) FROM account_transfers 
   WHERE to_account_id = ca.id) AS transfers_in,
  
  (SELECT COALESCE(SUM(amount), 0) FROM supplier_payments 
   WHERE cash_account_id = ca.id) AS supplier_payments,
  
  (SELECT COALESCE(SUM(amount), 0) FROM expenses 
   WHERE cash_account_id = ca.id) AS expenses,
  
  (SELECT COALESCE(SUM(amount), 0) FROM account_transfers 
   WHERE from_account_id = ca.id) AS transfers_out,
  
  (SELECT COALESCE(SUM(adjustment_amount), 0) 
   FROM balance_adjustments 
   WHERE cash_account_id = ca.id) AS adjustments,
  
  ca.opening_balance::numeric 
    + (SELECT COALESCE(SUM(amount),0) FROM customer_payments 
       WHERE cash_account_id = ca.id)
    + (SELECT COALESCE(SUM(amount),0) FROM payments 
       WHERE cash_account_id = ca.id AND status='completed')
