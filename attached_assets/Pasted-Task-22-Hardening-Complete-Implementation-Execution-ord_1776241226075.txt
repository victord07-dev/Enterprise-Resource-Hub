TASK 2 — Lot Engine: per-product margin + warehouse filter + logging
File: server/storage.ts (wherever computeFifoLots currently lives)
Replace the entire computeFifoLots function with the implementation below.
ts// ─── Types ────────────────────────────────────────────────────────────────────

export interface LotDetail {
  grnId: number;
  grnNumber: string;
  receivedDate: Date;
  remainingQty: number;
  landedCost: number;     // per-unit landed cost for this lot
  lotFloorPrice: number;  // landedCost × (1 + minMarginPct/100)
}

export interface FifoLotResult {
  lots: LotDetail[];                // only lots with remainingQty > 0
  blendedCost:  number | null;      // null when all stock depleted
  globalFloor:  number | null;      // min(lotFloorPrice) across active lots
  strictFloor:  number | null;      // max(lotFloorPrice) across active lots
  // engine meta (not in API response, used for logging)
  _meta: {
    productId:     number;
    warehouseId:   number | undefined;
    movementCount: number;
    lotCount:      number;
  };
}

// ─── Engine ───────────────────────────────────────────────────────────────────

/**
 * Compute FIFO lot breakdown for a product.
 *
 * @param productId     - required
 * @param warehouseId   - optional; when provided, filters stock_movements by warehouse_id
 * @param minMarginPct  - optional; pass the product's value to avoid a DB round-trip.
 *                        If omitted the engine fetches it from the products table.
 */
export async function computeFifoLots(
  productId: number,
  warehouseId?: number,
  minMarginPctOverride?: number,
): Promise<FifoLotResult> {

  // ── 1. Resolve per-product margin ─────────────────────────────────────────
  let marginPct: number;
  if (minMarginPctOverride !== undefined) {
    marginPct = minMarginPctOverride;
  } else {
    const [product] = await db
      .select({ minMarginPct: products.minMarginPct })
      .from(products)
      .where(eq(products.id, productId))
      .limit(1);
    marginPct = product ? Number(product.minMarginPct) : 5.0; // safe fallback
  }
  const marginMultiplier = 1 + marginPct / 100;

  // ── 2. Load GRN receipts for this product (FIFO order) ────────────────────
  const receipts = await db
    .select({
      grnId:        goodsReceiptNoteItems.grnId,
      grnNumber:    goodsReceiptNotes.grnNumber,
      receivedDate: goodsReceiptNotes.receivedDate,
      receivedQty:  goodsReceiptNoteItems.quantity,
      landedCost:   goodsReceiptNoteItems.landedCostPerUnit,  // adjust field name to match your schema
    })
    .from(goodsReceiptNoteItems)
    .innerJoin(goodsReceiptNotes, eq(goodsReceiptNoteItems.grnId, goodsReceiptNotes.id))
    .where(eq(goodsReceiptNoteItems.productId, productId))
    .orderBy(asc(goodsReceiptNotes.receivedDate), asc(goodsReceiptNoteItems.grnId));

  // ── 3. Load outbound movements (dispatch, adjustments) ───────────────────
  let movementsQuery = db
    .select({
      grnId:    stockMovements.grnId,
      qty:      stockMovements.quantity,
      type:     stockMovements.movementType,
    })
    .from(stockMovements)
    .where(
      and(
        eq(stockMovements.productId, productId),
        inArray(stockMovements.movementType, ["dispatch", "adjustment_out", "return_out"]),
      )
    );

  // Warehouse filter — applied only when warehouseId is provided
  if (warehouseId !== undefined) {
    movementsQuery = movementsQuery.where(
      and(
        eq(stockMovements.productId, productId),
        inArray(stockMovements.movementType, ["dispatch", "adjustment_out", "return_out"]),
        eq(stockMovements.warehouseId, warehouseId),
      )
    ) as typeof movementsQuery;
  }

  const movements = await movementsQuery;
  const movementCount = movements.length;

  // ── 4. Build consumed-qty map per GRN ────────────────────────────────────
  // For dispatch movements linked to a specific GRN lot (from the splitting done in task #22)
  const consumedByGrn = new Map<number, number>();
  let unlinkedConsumed = 0;

  for (const m of movements) {
    if (m.grnId) {
      consumedByGrn.set(m.grnId, (consumedByGrn.get(m.grnId) ?? 0) + Number(m.qty));
    } else {
      unlinkedConsumed += Number(m.qty);
    }
  }

  // ── 5. Apply FIFO to unlinked consumed qty ───────────────────────────────
  // Work through receipts oldest-first, draining unlinked consumption
  const receiptRemaining = new Map<number, number>();
  for (const r of receipts) {
    receiptRemaining.set(r.grnId, Number(r.receivedQty) - (consumedByGrn.get(r.grnId) ?? 0));
  }

  let toConsume = unlinkedConsumed;
  for (const r of receipts) {
    if (toConsume <= 0) break;
    const remaining = receiptRemaining.get(r.grnId) ?? 0;
    const drain = Math.min(remaining, toConsume);
    receiptRemaining.set(r.grnId, remaining - drain);
    toConsume -= drain;
  }

  // ── 6. Build active lot list ──────────────────────────────────────────────
  const activeLots: LotDetail[] = [];

  for (const r of receipts) {
    const remaining = receiptRemaining.get(r.grnId) ?? 0;
    if (remaining <= 0) continue;

    const landedCost = Number(r.landedCost);
    activeLots.push({
      grnId:         r.grnId,
      grnNumber:     r.grnNumber,
      receivedDate:  r.receivedDate,
      remainingQty:  remaining,
      landedCost,
      lotFloorPrice: parseFloat((landedCost * marginMultiplier).toFixed(4)),
    });
  }

  // ── 7. Compute aggregates ─────────────────────────────────────────────────
  let blendedCost:  number | null = null;
  let globalFloor:  number | null = null;
  let strictFloor:  number | null = null;

  if (activeLots.length > 0) {
    const totalQty   = activeLots.reduce((s, l) => s + l.remainingQty, 0);
    const totalValue = activeLots.reduce((s, l) => s + l.remainingQty * l.landedCost, 0);

    blendedCost = parseFloat((totalValue / totalQty).toFixed(4));
    globalFloor = parseFloat(Math.min(...activeLots.map(l => l.lotFloorPrice)).toFixed(4));
    strictFloor = parseFloat(Math.max(...activeLots.map(l => l.lotFloorPrice)).toFixed(4));
  }

  const result: FifoLotResult = {
    lots:        activeLots,
    blendedCost,
    globalFloor,
    strictFloor,
    _meta: {
      productId,
      warehouseId,
      movementCount,
      lotCount: activeLots.length,
    },
  };

  // ── 8. Mandatory logging ──────────────────────────────────────────────────
  console.log(
    `[LOT_ENGINE] productId=${productId} warehouseId=${warehouseId ?? "global"} ` +
    `movements=${movementCount} lots=${activeLots.length} ` +
    `blendedCost=${blendedCost ?? "null"} ` +
    `globalFloor=${globalFloor ?? "null"} strictFloor=${strictFloor ?? "null"}`,
  );

  if (process.env.DEBUG_LOT_ENGINE === "true") {
    for (const lot of activeLots) {
      console.log(
        `  [LOT_ENGINE_DEBUG] grnId=${lot.grnId} grnNumber=${lot.grnNumber} ` +
        `receivedDate=${lot.receivedDate.toISOString().slice(0, 10)} ` +
        `remainingQty=${lot.remainingQty} landedCost=${lot.landedCost} ` +
        `lotFloorPrice=${lot.lotFloorPrice}`,
      );
    }
  }

  return result;
}

TASK 3 — GET /api/inventory/stock-lot-summary endpoint
File: server/routes.ts
Add this route. Insert it near the other /api/inventory/ routes (around line 3480–3520 as noted in the spec):
ts// ─── GET /api/inventory/stock-lot-summary ────────────────────────────────────
app.get(
  "/api/inventory/stock-lot-summary",
  authenticateToken,
  async (req: Request, res: Response) => {
    try {
      const productIdRaw  = req.query.productId  as string | undefined;
      const warehouseIdRaw = req.query.warehouseId as string | undefined;

      // productId is required
      if (!productIdRaw) {
        return res.status(400).json({ error: "productId query parameter is required" });
      }

      const productId = parseInt(productIdRaw, 10);
      if (isNaN(productId)) {
        return res.status(400).json({ error: "productId must be a valid integer" });
      }

      const warehouseId = warehouseIdRaw ? parseInt(warehouseIdRaw, 10) : undefined;
      if (warehouseIdRaw && isNaN(warehouseId!)) {
        return res.status(400).json({ error: "warehouseId must be a valid integer" });
      }

      // Fetch minMarginPct upfront to pass into engine (avoids double DB hit)
      const [product] = await db
        .select({ minMarginPct: products.minMarginPct })
        .from(products)
        .where(eq(products.id, productId))
        .limit(1);

      if (!product) {
        return res.status(404).json({ error: "Product not found" });
      }

      const result = await computeFifoLots(
        productId,
        warehouseId,
        Number(product.minMarginPct),
      );

      // Shape the response — omit internal _meta
      const { _meta, ...publicResult } = result;

      return res.json({
        productId,
        warehouseId: warehouseId ?? null,
        lots:        publicResult.lots,          // [] when all stock depleted
        blendedCost: publicResult.blendedCost,   // null when all stock depleted
        globalFloor: publicResult.globalFloor,   // null when all stock depleted
        strictFloor: publicResult.strictFloor,   // null when all stock depleted
      });
    } catch (err) {
      console.error("[LOT_SUMMARY_API] Error:", err);
      return res.status(500).json({ error: "Internal server error" });
    }
  },
);
